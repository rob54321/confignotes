How to pass parameters from bash to inline perl and back to bash.

#!/bin/bash
get () {
	ANSWER=$(perl -e 'my $nam = shift; print "hello $nam";' "$1")
	echo "${ANSWER}";
}

LIST=`get "Robert"`
echo "list: ${LIST}"

#!/bin/bash
get () {
	ANSWER=$(perl -e 'my $nam = shift;
			  my $sur = shift;
	                  print "hello $nam $sur";' "$1" "$2")
	echo "${ANSWER}";
}

LIST=`get "Robert Key"`
echo "list: ${LIST}"

#!/bin/bash

# calling a perl sub from bash
# parameters passed 1 and 2 strings
# return status and a string
perlsub() {
	# silly perl sub to called from bash
	RESULT=$(perl -e 'use strict;
					  use warnings;
					  my $param1 = shift;
					  my $param2 = shift;
					  my $result;
					  # do something
					  $result = $param1 . $param2;
					  # return value to bash
					  print "$result";' "$1" "$2")
	# echo "end of perlsub"

	# return status
	if test "${1}" = "hello"; then
		# this return status could mean success
		return 0;
	else
		# this return status could mean failure
		return 1;
	fi

}

# call perlsub
perlsub "hel" "hello"
rc="$?"

echo "status = ${rc}"
echo "return string ${RESULT}"
