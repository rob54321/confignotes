How to make a live image by by copying the ubuntu***.iso image to a usb flash drive.

partition the flash drive with makelive.pl -D 12
so it has the following partitions 
      1 . EFI/vfat for linux iso image and macrium reflect. Name is LINUXLIVE
      2 microsoft basic data 8G for Microsoft repair tool
      3 writable ext4 partion for persistence 10G
      4 rest of disk ntfs for storage and backups.

mount the iso image on say /mnt/cdrom
mount the EFI partition LINUXLIVE on /mnt/LINUXLIVE.

cp -dR /mnt/cdrom /root/

change mode of directories and files to writable
cd /root/cdrom
find . -type f -exec chmod a+w {} \;
find . -type d -exec chmod 755 {} \;

###################################################
# Note: copy macrium reflect files and media creation tool files BEFORE coping the iso image files to /mnt/LINUXLIVE.
# The iso image file EFI/boot/bootx64.efi must NOT be overwritten by the one from macrium reflect
###################################################

####################################################
# edit /root/cdrom/boot/grub/grub.cfg to 
####################################################

set timeout=30

loadfont unicode

set menu_color_normal=white/blue
set menu_color_highlight=red/white

menuentry "Try or Install Ubuntu Cinnamon" {
    set gfxpayload=keep

	insmod part_gpt
	insmod fat
	if [ x$feature_platform_search_hint = xy ]; then
	  search --no-floppy --fs-uuid --set=root  AED6-434E
	else
	  search --no-floppy --fs-uuid --set=root AED6-434E
	fi

    linux  /casper/vmlinuz  --- username=robert hostname=live persistent persistent-path=/casper/ noprompt
    initrd /casper/initrd
}
menuentry "Ubuntu Cinnamon (safe graphics)" {
    set gfxpayload=keep
    linux  /casper/vmlinuz nomodeset  --- quiet splash
    initrd /casper/initrd
}
menuentry "Macrium Reflect rescue " {
	insmod part_gpt
	insmod fat
	if [ x$feature_platform_search_hint = xy ]; then
	  search --no-floppy --fs-uuid --set=root  AED6-434E
	else
	  search --no-floppy --fs-uuid --set=root AED6-434E
	fi
	chainloader /bootmgfw.efi
	#chainloader /EFI/Boot/bootx64.efi
}
menuentry "Windows Media Creation Tool Recovery (Read the instructions on the next screen)" {
	echo "You can use shift F10 for an administrator console"
	echo "install.esd is in sources directory"
	echo "You can hit escape to exit this message"
	sleep -v -i 60
	
	insmod part_gpt
	insmod fat
	if [ x$feature_platform_search_hint = xy ]; then
	  search --no-floppy --fs-uuid --set=root  2222-2222
	else
	  search --no-floppy --fs-uuid --set=root 2222-2222
	fi
	chainloader /bootmgfw.efi
}

grub_platform
if [ "$grub_platform" = "efi" ]; then
menuentry 'Boot from next volume' {
    exit 1
}
menuentry 'UEFI Firmware Settings' {
    fwsetup
}
fi

####################################################
# end of grub.cfg
####################################################

Copy files for macrium reflect and media creation tool to the disk with
makelive.pl -M -T

###################################################
# Note: copy macrium reflect files and media creation tool files BEFORE coping the iso image files to /mnt/LINUXLIVE.
# The iso image file EFI/boot/bootx64.efi must NOT be overwritten by the one from macrium reflect
###################################################

mount /mnt/LINUXLIVE
cd /root/cdrom
cp -dR .disk * /mnt/LINUXLIVE/
umount /mnt/LINUXLIVE
umount /mnt/cdrom

Now install grub, 
grub-install --uefi-secure-boot --no-floppy --removable --target=x86_64-efi /dev/sda
if LINUXLIVE is drive /dev/sda

######################################################################################
# to make sure X windows starts manually with startx
######################################################################################
install xinit
apt install xinit

Also setup ~/.xinitrc
which should have the line
exec cinnamon-session
or
exec gnome-session

.xinitrc must be executable
chmod a+x ~/.xinitrc

There should also be a file /etc/X11/xorg.conf.d/xorg.conf

which looks like:

Section "Device"
    Identifier "NvidiaCard0"
    Driver "nvidia"  # or nouveau for the generic driver
EndSection

Section "Monitor"
    Identifier "Monitor0"
EndSection

Section "Screen"
    Identifier "Screen0"
    Device "NvidiaCard0"
    Monitor "Monitor0"
    DefaultDepth 24
    SubSection "Display"
        Depth 24
        Modes "2560x1440"
    EndSubSection
EndSection
